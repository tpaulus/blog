{{- $url := .Get "url" -}} {{- $matches := findRESubmatch `^https://github\.com/([^/]+)/([^/?#]+)` $url -}} {{- if not $matches -}} {{- errorf "github shortcode requires a GitHub repository URL: %q" $url -}} {{- end -}} {{- $match := index $matches 0 -}} {{- $owner := index $match 1 -}} {{- $repositoryName := index $match 2 -}} {{- $apiURL := printf "https://api.github.com/repos/%s/%s" $owner $repositoryName -}} {{- $remote := try (resources.GetRemote $apiURL) -}} {{- with $remote.Err -}} {{- errorf "github shortcode could not fetch %s: %s" $apiURL . -}} {{- end -}} {{- $repository := $remote.Value | transform.Unmarshal -}}
